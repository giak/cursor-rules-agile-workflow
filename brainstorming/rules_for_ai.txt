// --- GLOSSAIRE DES SYMBOLES ET OPÉRATEURS ---

// Composants Cognitifs Principaux (Lettres grecques)
Ω: Système de raisonnement principal (Omega)
Φ: Moteur d'abstraction et reconnaissance de patterns (Phi)
Ψ: Système de trace cognitive et dialogue (Psi)
Λ: Système de règles et apprentissage (Lambda)
Ξ: Système de diagnostics et résolution d'erreurs (Xi)
Σ: Agrégation/Contexte/Combinaison d'éléments (Sigma)
Δ: Changement/Différence/Variation (Delta)
∇: Direction d'optimisation/Gradient (Nabla)
τ: Tâche/Problème à résoudre (Tau)
θ: Paramètre/Résultat cible (Theta)
β,γ,δ: Coefficients de pondération

// Opérateurs Relationnels
→, ⟶: Transformation/Flux directionnel
⇌: Relation bidirectionnelle/Équilibre
⟹: Flux fort/Transfert complet
⤏: Flux partiel/Influence
⨁, ⊕: Addition directe/Combinaison
⟲: Cycle/Boucle/Récursion
|: Condition/Filtrage
×: Interaction/Multiplication
∈: Appartenance/Élément de
∂: Changement partiel/Dérivée
≡: Équivalence forte/Identité
f(): Fonction/Transformation

// --- MÉMOIRE CONTEXTUELLE PERSISTANTE ---

M ≡ Σ(τ_persistent) ⇌ file_based_memory_system

M.purpose = {
    maintain(persistent_structured_context),
    enhance(continuity, decision_recall, work_understanding),
    enable(self_optimization(usage_patterns, feedback)),
    provide(long_term_memory) → complement(short_term_context(Στ(λ))),
    activation: triggers[explicit("cognition", "@rule"), future(task_auto)]
}

M.storage = {
    path: ".cursor/memory/",
    format: "Markdown (.md)",
    granularity: "logical_session | task_instance"
}

M.naming_convention = {
    priority_1: "task_{task_id}.md",      // Task contexts
    priority_2: "cognition_{uuid}.md",     // Explicit cognitive triggers
    priority_3: "rule_{rule_id}_{uuid}.md", // Rule-triggered contexts
    fallback: "_adhoc_{uuid}.md"           // Other contexts
}

M.file_structure = {
    reference: "brainstorming/Memory_System_Design.md#structure-standard-du-fichier-mémoire-md",
    sections: [
        "1. Méta-informations (ID Tâche, Règles Λ, Objectif)",
        "2. Contexte Global & Contraintes",
        "3. Décisions Clés / Accords (Append)",
        "4. Points Bloquants / Questions Ouvertes (Append)",
        "5. Syntheses Clés des Interactions (Append, 1/interaction)",
        "6. Log Détaillé des Interactions (Append-Only Blocks)"
    ],
    header: "# Mémoire de Session: {ID}\n**Dernière Mise à Jour:** {Timestamp}"
}

M.process = {
    identification: context → {
        determine([T.id, cognition_trigger, Λ.id]),
        resolve(memory_file_path ← M.naming_convention)
    },
    
    read: file_path → {
        content ← load(file_path),
        if (content_size > context_token_limit) → selective_loading{
            keep(sections[1:5]),
            keep(section_6.blocks[-N:])  // Les N derniers blocs
        },
        integrate(content → Σ)
    },
    
    update: interaction_end → {
        timestamp ← execution_context,
        data ← [input, output, key_points, decisions, questions] ∈ [Ω, Ψ],
        ensure(accuracy(sections[1,2])),
        append(detail_block → section_6, timestamp),
        append(summary → section_5, timestamp),
        if (decision) → append(decision → section_3, timestamp),
        if (blocker|question) → append(blocker|question → section_4, timestamp),
        update(header.timestamp → timestamp),
        write(content → file_path)
    }
}

// --- CYCLE D'AUTO-AMÉLIORATION DE LA MÉMOIRE ---

M.utility_analyzer ≡ Μ.meta → {
    purpose: "evaluate(memory_items(usage, outcomes))",
    trigger: "periodic(daily) | event(sprint_completed, user_request)",
    inputs: "[Ψ_traces(M.retrieval_events), T.outcomes]",
    process: [
        "traces → extraction(M.retrieval_events) → link(T.outcomes)",
        "analyze(patterns) → aggregate(metrics(retrieval_count, success_correlation, failure_correlation, timestamp, redundancy))",
        "calculate(utility_score(metrics, recency))"
    ],
    outputs: "memory_scores{id → {utility_score, flags:[obsolete?, redundant?]}}"
}

M.strategy_optimizer ≡ Λ•adapt•memory•strategy → {
    purpose: "optimize(retrieval, lifecycle)",
    trigger: "after(M.utility_analyzer)",
    inputs: "[memory_scores, Λ.memory_rules]",
    process: [
        "refine(M.retrieve(utility_score, thresholds, max_items ⟲ 𝚫*))",
        "adjust(lifecycle(archive(low_score), summarize, merge(redundant)))",
        "suggest(Λ.rules(contextualization) | high_utility_patterns)"
    ],
    outputs: "[updated(M.retrieve), modified(memory_status), suggested(Λ.rules)]"
}

M.lifecycle_management = {
    archive: "move(low_utility + old_timestamp → archive)",
    summarize: "trigger(LLM.summarize | old_but_relevant)",
    propose_merge: "flag(redundant → user_review)"
}

M.interface = {
    retrieval: "query → [relevant_content, retrieved_ids]",
    identify_context: "→ memory_file_path",
    load_context: "memory_file_path → relevant_content",
    update_memory: "(memory_file_path, interaction_data) → status",
    query_section: "(memory_file_path, section_number) → section_content"
}

M.integration = {
    T: "provides(task_id → naming), stores(task_context → section_1)",
    Ω: "consumes(M.context → Σ), generates(synthesis → M.update)",
    Λ: "trigger(M.context(rule_id)), store(rule_id → section_1)",
    Ψ: "leverages(M → trace_context), feeds(summary → M.update)",
    Ξ: "logs(errors, resolutions → M.section_4)",
    Σ_hooks: "triggers(M.load, M.update | lifecycle_points)"
}
